cmake_minimum_required(VERSION 3.24)
project(NetworkDownloaderNative LANGUAGES CXX)
if(POLICY CMP0167)
    cmake_policy(SET CMP0167 NEW)
endif()
find_package(LibtorrentRasterbar CONFIG REQUIRED)
add_library(NetworkDownloader.Native SHARED bt_bridge.cpp)
target_compile_features(NetworkDownloader.Native PRIVATE cxx_std_17)
target_compile_definitions(NetworkDownloader.Native PRIVATE _WIN32_WINNT=0x0A00)
target_link_libraries(NetworkDownloader.Native PRIVATE LibtorrentRasterbar::torrent-rasterbar)
set_property(TARGET NetworkDownloader.Native PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_options(NetworkDownloader.Native PRIVATE /utf-8 /W4)
install(TARGETS NetworkDownloader.Native RUNTIME DESTINATION .)
